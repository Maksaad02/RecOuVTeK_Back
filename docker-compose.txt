version: '3.8'

services:
  mysql:
    image: mysql:8.0
    container_name: recouvtek_mysql
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: Maksaad
      MYSQL_DATABASE: rec_ouv_db
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql
      - ./docker/init.sql:/docker-entrypoint-initdb.d/init.sql:ro
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "-uroot", "-pMaksaad"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    command: --default-authentication-plugin=mysql_native_password

volumes:
  mysql_data:
    driver: local
